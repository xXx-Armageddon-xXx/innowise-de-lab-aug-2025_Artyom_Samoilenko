CREATE TABLE "Customers" (
  "CustomerID" INTEGER,
  "FirstName" VARCHAR(50),
  "LastName" VARCHAR(50),
  "Phone" VARCHAR(15),
  PRIMARY KEY ("CustomerID")
);

CREATE TABLE "Reservations" (
  "CustomerID" INTEGER,
  "ReservationID" INTEGER,
  "ReservationDate" DATE,
  "ReservationStart" TIME,
  "ReservationEnd" Time,
  PRIMARY KEY ("ReservationID"),
  CONSTRAINT "FK_Reservations_ReservationID"
    FOREIGN KEY ("ReservationID")
      REFERENCES "Customers"("FirstName")
);

CREATE TABLE "Staff" (
  "TablesID" INTEGER,
  "StaffID" INTEGER,
  "StaffName" VARCHAR(50),
  PRIMARY KEY ("StaffID"),
  CONSTRAINT "FK_Staff_StaffID"
    FOREIGN KEY ("StaffID")
      REFERENCES "Reservations"("ReservationID")
);

CREATE TABLE "Tables" (
  "TableID" INTEGER,
  "IsActive" BOOLEAN,
  "Capacity" INTEGER,
  "StaffID" INTEGER,
  PRIMARY KEY ("TableID"),
  CONSTRAINT "FK_Tables_IsActive"
    FOREIGN KEY ("IsActive")
      REFERENCES "Staff"("StaffName")
);

CREATE TABLE "ReservationsTables" (
  "ReservationID" Тип,
  "TableID" Тип
);
